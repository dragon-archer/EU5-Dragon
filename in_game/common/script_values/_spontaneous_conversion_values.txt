sc_conversion_final_scale = {
	value = population # in 1k units
	multiply = {
		value = local_religious_unity
		multiply = {
			value = 1
			subtract = {
				value = local_religious_unity
				multiply = 0.75
			}
		}
		multiply = 3 # so that at 0.67 unity it's 1x speed
	}

	divide = {
		# value = -0.6 # compensate for initial scale
		every_present_religion_in_location = {
			add = 0.5
		}
	}

	if = {
		limit = { location_rank = location_rank:rural_settlement }
		multiply = 0.5
	}
	else_if = {
		limit = { location_rank = location_rank:town }
		multiply = 0.75
	}

	multiply = 0.002 # per month base speed
	min = 0.001
}
