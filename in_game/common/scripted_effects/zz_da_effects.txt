REPLACE:enforce_culture_effect = {
	$target$ = {
		change_culture = $actor$.culture
		add_cooldown = {
			type = convert_culture
			years = 2 # 10
		}
		every_cabinet = {
			limit = {
				has_cabinet_action = yes
				or = {
					cabinet_action = cabinet_action:assimilate_area
					cabinet_action = cabinet_action:promote_culture
				}
			}
			stop_cabinet_action = {}
		}
		capital = {
			every_pop = {
				limit = {
					NOT = {
						culture = $actor$.culture
					}
				}
				split_pop = {
					fraction = 0.16
					culture = $actor$.culture
				}
			}
		}
	}
}

# Input: var, variable name
da_sqrt = {
	if = {
		limit = {
			var:$var$ > 0
		}
		set_local_variable = {
			name = _da_sqrt_scale
			value = 1
		}
		set_local_variable = {
			name = _da_sqrt_in
			value = var:$var$
		}
		set_local_variable = {
			name = _da_sqrt_out
			value = 2
		}
		set_local_variable = {
			name = _da_sqrt_cnt
			value = 0
		}
		while = {
			limit = {
				local_var:_da_sqrt_in >= 4
			}
			change_local_variable = {
				name = _da_sqrt_scale
				multiply = 2
			}
			change_local_variable = {
				name = _da_sqrt_in
				multiply = 0.25
			}
		}
		while = {
			limit = {
				local_var:_da_sqrt_in < 1
			}
			change_local_variable = {
				name = _da_sqrt_scale
				multiply = 0.5
			}
			change_local_variable = {
				name = _da_sqrt_in
				multiply = 4
			}
		}
		while = {
			limit = {
				local_var:_da_sqrt_cnt < 10
			}
			change_local_variable = {
				name = _da_sqrt_cnt
				add = 1
			}
			set_local_variable = {
				name = _da_sqrt_out
				value = {
					value = local_var:_da_sqrt_out
					add = {
						value = local_var:_da_sqrt_in
						divide = local_var:_da_sqrt_out
					}
					multiply = 0.5
				}
			}
		}
		set_variable = {
			name = $var$
			value = {
				value = local_var:_da_sqrt_out
				multiply = local_var:_da_sqrt_scale
			}
		}
		remove_local_variable = _da_sqrt_scale
		remove_local_variable = _da_sqrt_in
		remove_local_variable = _da_sqrt_out
		remove_local_variable = _da_sqrt_cnt
	}
}
